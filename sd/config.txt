#
# config.txt - BM-DOS
#
# Based on Circle's boot/config64.txt.  The firmware reads this before the
# emulator runs; it is not a Circle application file.
#
# Note that this file selects the kernel per Raspberry Pi model, so a single
# card can carry a Pi 3, 4 and 5 build at once if you copy all three images.
#

# 64-bit mode.  Circle's AArch64 port needs this and the kernel load address.
arm_64bit=1
kernel_address=0x80000

# initial_turbo stays at 0, although it is the usual trick for a faster boot:
# while it lasts, the firmware keeps the ARM at its top clock and answers
# clock requests without obeying them, and this port's governor takes that
# for a firmware that refuses clock changes and switches itself off
# (src/platform/bare_sysinfo.cpp).
initial_turbo=0

# A faster, cleaner boot: no rainbow square, and no pause before the kernel is
# loaded (boot_delay's default is 1 second).  The BM-DOS logo comes up as soon
# as the kernel runs.
disable_splash=1
boot_delay=0

# Keep the GPU footprint small: the emulator is CPU bound and we only need a
# framebuffer, not 3D or codecs.  64 MB is plenty for 1080p and leaves the rest
# of the RAM to DOSBox.
gpu_mem=64

# HDMI: force a mode instead of letting the firmware pick, so the frame buffer
# Circle allocates always matches what the display does.  1920x1080@60 is the
# safest choice for a TV or a modern monitor.  Change both lines together with
# DisplayWidth/DisplayHeight in src/kernel.cpp.
hdmi_group=1
hdmi_mode=16
hdmi_drive=2

# Overscan.  This is not cosmetic and it was the 1824x984.
#
# In CEA modes (hdmi_group=1, i.e. what a TV uses) the firmware defaults to a
# 48-pixel overscan border on all four edges, and the frame buffer it allocates
# is "the display's size minus the overscan" - that is the wording in the
# firmware's own config.txt template.  On 1080p that is 1920-96 = 1824 wide and
# 1080-96 = 984 tall, which is the size this port printed on the board
# ("Circle's console is 1824x984") when cmdline.txt did not ask for a mode.
#
# Disabling it gives the display the whole 1920x1080 and the emulator the whole
# screen.  If your TV then crops the outer few pixels of the picture (some sets
# overscan no matter what the source says), the other way round is to leave
# overscan alone and set the four edges explicitly, e.g.
#   overscan_left=0  overscan_right=0  overscan_top=0  overscan_bottom=0
# or a small negative value to zoom out.  It has to be *before* any [section]
# line: the firmware applies this globally.
disable_overscan=1

# Sound.  The emulator drives the SoC's audio hardware directly (Circle's HDMI
# or PWM device), so the firmware needs the audio subsystem enabled - that is
# what dtparam=audio=on does - but it is *not* the firmware that plays anything.
#
# If the display has speakers and the panel says "audio HDMI", the two lines
# that matter are the HDMI ones: hdmi_drive=2 is what tells the firmware to
# send audio over HDMI (drive=1 is DVI, video only).  It is already set above.
#
# If there is no sound over HDMI but there is over the jack, that is the
# display, not the port: run "make AUDIO_DEVICE=pwm" and the port will open the
# 3.5 mm jack instead.
dtparam=audio=on

# CPU frequency: nothing is set here on purpose.
#
# The port's own governor sets the ARM clock rate at run time, from 600 MHz at
# the DOS prompt up to the rate the emulator actually needs (see
# src/platform/bare_sysinfo.cpp).  Pinning arm_freq here would take that decision
# away from it - the firmware would report a maximum equal to the pinned value
# and tie every level to the same number.
#
# If your board is unstable and you want it slower, *that* is when to put a line
# here, e.g. arm_freq=1500, and the governor will build its levels inside
# whatever range the firmware then reports.

# Two frame buffers on the Pi 4/5: Circle uses the second one for page flipping,
# which is what makes the scaling tear free.
max_framebuffers=2

[pi02]
kernel=kernel8.img

[pi2]
kernel=kernel8.img

[pi3]
kernel=kernel8.img

[pi3+]
kernel=kernel8.img

[pi4]
# The Circle ARM stub is required for FIQ support on a Pi 4.  Build it with
#   make -C third_party/circle-stdlib/libs/circle/boot armstub64
# and copy armstub8-rpi4.bin next to this file.  If you skip it, remove the
# armstub line - the port does not use FIQ, but Circle's default config for
# the Pi 4 expects the file to exist.
armstub=armstub8-rpi4.bin
